import 'package:flutter/material.dart';
import 'package:flutter/services.dart'; // <-- needed for SystemNavigator.pop()

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: ParentPage(),
    );
  }
}

/// PARENT WIDGET (Stateful):
/// - Owns the state (counter)
/// - Contains the functions that change the state
/// - Sends state DOWN to children
/// - Receives events UP from children using callbacks
class ParentPage extends StatefulWidget {
  const ParentPage({super.key});

  @override
  State<ParentPage> createState() => _ParentPageState();
}

class _ParentPageState extends State<ParentPage> {
  // ✅ This is the STATE (data that changes)
  int counter = 0;

  // ✅ Parent modifies state using setState()
  void increaseCounter() {
    setState(() {
      counter++;
    });
  }

  void decreaseCounter() {
    setState(() {
      counter--;
    });
  }

  @override
  Widget build(BuildContext context) {
    // Parent passes state + callbacks to children
    return Scaffold(
      appBar: AppBar(title: const Text("Simple State Demo")),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            // CHILD: only displays the state
            CounterText(value: counter),

            const SizedBox(height: 20),

            // CHILDREN: buttons that trigger parent logic
            Row(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                ActionButton(
                  text: "Increase",
                  onPressed: increaseCounter,
                ),
                const SizedBox(width: 12),
                ActionButton(
                  text: "Decrease",
                  onPressed: decreaseCounter,
                ),
              ],
            ),

            const SizedBox(height: 20),

            // NEW CHILD: Exit App button as a separate widget
            const ExitAppButton(),
          ],
        ),
      ),
    );
  }
}

/// CHILD 1 (Stateless):
/// - Receives counter value from parent
/// - Displays it (no state changes here)
class CounterText extends StatelessWidget {
  final int value;

  const CounterText({super.key, required this.value});

  @override
  Widget build(BuildContext context) {
    return Text(
      "Counter = $value",
      style: const TextStyle(fontSize: 28, fontWeight: FontWeight.bold),
    );
  }
}

/// CHILD 2 (Stateless):
/// - Reusable button widget
/// - Receives a callback from parent
///
/// VoidCallback means: void Function()
/// -> a function that takes no parameters and returns nothing.
class ActionButton extends StatelessWidget {
  final String text;
  final VoidCallback onPressed; // stores a function reference

  const ActionButton({
    super.key,
    required this.text,
    required this.onPressed,
  });

  @override
  Widget build(BuildContext context) {
    return ElevatedButton(
      // IMPORTANT:
      // onPressed: onPressed   -> passes the function reference
      // onPressed: onPressed() -> would CALL it immediately (wrong)
      onPressed: onPressed,
      child: Text(text),
    );
  }
}

/// CHILD 3 (Stateless):
/// - A separate widget that closes the application
/// - Uses SystemNavigator.pop() (works mainly on Android)
///
/// Educational note:
/// In real apps, you usually DON'T force-close the app.
/// Users close apps themselves.
class ExitAppButton extends StatelessWidget {
  const ExitAppButton({super.key});

  void exitApp() {
    // Closes the app (Android). On iOS this is generally not allowed.
    SystemNavigator.pop();
  }

  @override
  Widget build(BuildContext context) {
    return OutlinedButton.icon(
      onPressed: exitApp,
      icon: const Icon(Icons.exit_to_app),
      label: const Text("Exit App"),
    );
  }
}
