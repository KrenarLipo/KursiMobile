import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      title: 'Color Mix State Demo',
      theme: ThemeData(useMaterial3: true),
      home: const ColorMixScreen(),
    );
  }
}

class ColorMixScreen extends StatefulWidget {
  const ColorMixScreen({super.key});

  @override
  State<ColorMixScreen> createState() => _ColorMixScreenState();
}

class _ColorMixScreenState extends State<ColorMixScreen> {
  // Only 5 allowed colors
  final Map<String, Color> colors = const {
    'Red': Colors.red,
    'Green': Colors.green,
    'Blue': Colors.blue,
    'Black': Colors.black,
    'White': Colors.white,
  };

  // ----------------------------
  // MAIN CIRCLE STATE
  // ----------------------------
  Color mainColor = Colors.grey; // default
  String mainLabel = 'Default';

  // ----------------------------
  // MIXER STATE
  // ----------------------------
  String? colorAName;
  String? colorBName;
  Color? colorA;
  Color? colorB;

  // Linked percentages: A% + B% = 100
  double aPercent = 50; // 0..100
  double bPercent = 50; // 0..100

  // LOADING STATE (new)
  bool isApplying = false;

  // SCROLL CONTROLLER (new)
  final ScrollController _scrollController = ScrollController();

  // Convert percentage to lerp parameter:
  // Color.lerp(A, B, t) where t=0 => 100% A, t=1 => 100% B
  double get t => bPercent / 100.0;

  bool get canMix => colorA != null && colorB != null;
  bool get canApply => canMix && !isApplying;

  Color? get mixedColor {
    if (!canMix) return null;
    return Color.lerp(colorA, colorB, t);
  }

  // ----------------------------
  // Linked slider logic
  // ----------------------------
  void setAPercent(double newA) {
    setState(() {
      aPercent = newA.clamp(0, 100);
      bPercent = 100 - aPercent;
    });
  }

  void setBPercent(double newB) {
    setState(() {
      bPercent = newB.clamp(0, 100);
      aPercent = 100 - bPercent;
    });
  }

  void selectColorA(String name) {
    setState(() {
      colorAName = name;
      colorA = colors[name]!;
    });
  }

  void selectColorB(String name) {
    setState(() {
      colorBName = name;
      colorB = colors[name]!;
    });
  }

  // ----------------------------
  // APPLY: show loading, then update, then scroll to top
  // ----------------------------
  Future<void> applyMixToMainCircle() async {
    final mix = mixedColor;
    if (mix == null) return;

    // 1) Show loading spinner
    setState(() {
      isApplying = true;
    });

    // Simulate a small "loading" delay (like network / heavy work)
    await Future.delayed(const Duration(milliseconds: 700));

    if (!mounted) return;

    // 2) Apply changes to main circle
    setState(() {
      mainColor = mix;

      final a = colorAName!;
      final b = colorBName!;
      mainLabel = 'Mix: ${aPercent.round()}% $a + ${bPercent.round()}% $b';

      isApplying = false;
    });

    // 3) Scroll to top automatically to see the updated main circle
    //    (only if the scroll controller is attached)
    if (_scrollController.hasClients) {
      await _scrollController.animateTo(
        0,
        duration: const Duration(milliseconds: 450),
        curve: Curves.easeOut,
      );
    }
  }

  @override
  void dispose() {
    // Always dispose controllers to avoid memory leaks
    _scrollController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final preview = mixedColor;

    return Scaffold(
      appBar: AppBar(
        title: const Text('State Interaction (Apply + Loading + Auto-Scroll)'),
        centerTitle: true,
      ),
      body: SingleChildScrollView(
        controller: _scrollController, // (new) enables scrolling programmatically
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            // 1) MAIN CIRCLE (only changes after Apply)
            Card(
              elevation: 2,
              child: Padding(
                padding: const EdgeInsets.all(16),
                child: Column(
                  children: [
                    const Text(
                      'Main Circle (changes ONLY after Apply)',
                      style: TextStyle(fontSize: 16, fontWeight: FontWeight.w600),
                    ),
                    const SizedBox(height: 14),
                    CircleAvatar(radius: 70, backgroundColor: mainColor),
                    const SizedBox(height: 12),
                    Text(
                      mainLabel,
                      textAlign: TextAlign.center,
                      style: const TextStyle(fontSize: 16),
                    ),
                  ],
                ),
              ),
            ),

            const SizedBox(height: 18),

            // 2) Select Color A
            const Text(
              'Step 1: Select Color A',
              style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 10),
            Wrap(
              spacing: 10,
              runSpacing: 10,
              children: colors.entries.map((entry) {
                return ChoiceChip(
                  label: Text(entry.key),
                  selected: entry.key == colorAName,
                  avatar: CircleAvatar(backgroundColor: entry.value),
                  onSelected: (_) => selectColorA(entry.key),
                );
              }).toList(),
            ),
            const SizedBox(height: 8),
            Text('Color A: ${colorAName ?? "Not selected"}'),

            const SizedBox(height: 18),

            // 3) Select Color B
            const Text(
              'Step 2: Select Color B',
              style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 10),
            Wrap(
              spacing: 10,
              runSpacing: 10,
              children: colors.entries.map((entry) {
                return ChoiceChip(
                  label: Text(entry.key),
                  selected: entry.key == colorBName,
                  avatar: CircleAvatar(backgroundColor: entry.value),
                  onSelected: (_) => selectColorB(entry.key),
                );
              }).toList(),
            ),
            const SizedBox(height: 8),
            Text('Color B: ${colorBName ?? "Not selected"}'),

            const SizedBox(height: 22),
            const Divider(),
            const SizedBox(height: 12),

            // 4) Linked scrollbars (sliders)
            const Text(
              'Step 3: Mix Level (Linked Scrollbars)',
              style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 10),

            _percentRow(label: 'Color A %', value: aPercent),
            Slider(
              value: aPercent,
              min: 0,
              max: 100,
              divisions: 100,
              label: '${aPercent.round()}%',
              onChanged: isApplying ? null : setAPercent, // disable while loading
            ),

            const SizedBox(height: 10),

            _percentRow(label: 'Color B %', value: bPercent),
            Slider(
              value: bPercent,
              min: 0,
              max: 100,
              divisions: 100,
              label: '${bPercent.round()}%',
              onChanged: isApplying ? null : setBPercent, // disable while loading
            ),

            const SizedBox(height: 16),

            // Mix preview
            Card(
              elevation: 1,
              child: Padding(
                padding: const EdgeInsets.all(14),
                child: Row(
                  children: [
                    Expanded(
                      child: Column(
                        children: [
                          const Text('A', style: TextStyle(fontWeight: FontWeight.bold)),
                          const SizedBox(height: 8),
                          CircleAvatar(
                            radius: 22,
                            backgroundColor: colorA ?? Colors.grey.shade300,
                          ),
                          const SizedBox(height: 6),
                          Text('${aPercent.round()}%'),
                        ],
                      ),
                    ),
                    const Icon(Icons.add),
                    Expanded(
                      child: Column(
                        children: [
                          const Text('B', style: TextStyle(fontWeight: FontWeight.bold)),
                          const SizedBox(height: 8),
                          CircleAvatar(
                            radius: 22,
                            backgroundColor: colorB ?? Colors.grey.shade300,
                          ),
                          const SizedBox(height: 6),
                          Text('${bPercent.round()}%'),
                        ],
                      ),
                    ),
                    const Icon(Icons.arrow_forward),
                    Expanded(
                      child: Column(
                        children: [
                          const Text('Preview', style: TextStyle(fontWeight: FontWeight.bold)),
                          const SizedBox(height: 8),
                          CircleAvatar(
                            radius: 22,
                            backgroundColor: preview ?? Colors.grey.shade300,
                          ),
                          const SizedBox(height: 6),
                          Text(canMix ? 'Mixed' : 'Select A & B'),
                        ],
                      ),
                    ),
                  ],
                ),
              ),
            ),

            const SizedBox(height: 16),

            // APPLY button with loading spinner (new)
            ElevatedButton(
              onPressed: canApply ? applyMixToMainCircle : null,
              child: Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  if (isApplying) ...[
                    const SizedBox(
                      width: 18,
                      height: 18,
                      child: CircularProgressIndicator(strokeWidth: 2),
                    ),
                    const SizedBox(width: 10),
                    const Text('Applying...'),
                  ] else ...[
                    const Icon(Icons.check_circle),
                    const SizedBox(width: 10),
                    const Text('Apply Mix to Main Circle'),
                  ],
                ],
              ),
            ),

            const SizedBox(height: 8),
            Text(
              canMix
                  ? (isApplying ? 'Please wait...' : 'Ready to apply.')
                  : 'Select Color A and Color B to enable Apply.',
              textAlign: TextAlign.center,
              style: TextStyle(
                fontStyle: FontStyle.italic,
                color: canMix ? null : Theme.of(context).colorScheme.error,
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _percentRow({required String label, required double value}) {
    return Row(
      children: [
        Expanded(
          child: Text(label, style: const TextStyle(fontWeight: FontWeight.w600)),
        ),
        Text('${value.round()}%'),
      ],
    );
  }
}
